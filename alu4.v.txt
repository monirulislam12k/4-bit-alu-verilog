module alu4 (
input wire [3:0] a,
input wire [3:0] b,
input wire [2:0] sel,
output reg [3:0] y,
output reg carry
);

always @(*)begin
y = 4'b0000;
carry = 1'b0;

case (sel)
3'b000: {carry,y} = {1'b0,a} + {1'b0,b};
3'b001: y = a - b;
3'b010: y = a & b;
3'b011: y = a | b;
3'b100: y = a ^ b;
3'b101: y = ~a;
3'b110: y = a << 1;
3'b111: y = a >> 1;
default: begin
y = 4'b0000;
carry = 1'b0;
end 
endcase
end 
endmodule
